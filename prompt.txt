## PHASE 1 — CUSTOM SERVER + SOCKET FOUNDATION
Files: server.js, lib/simulator.js

### server.js
- Express app on port 3001
- Attach Socket.IO to the Express server
- On 'connection': log the socket id, send current fleet state immediately
  via socket.emit('fleet:state', getAllShips())
- Import and start the simulator tick after server is ready
- Proxy Next.js on port 3000 (run Next.js separately in dev)

### lib/simulator.js
- Load fleet.json on startup
- Export: getAllShips(), getShip(id), updateShip(id, patch)
- startTick(): setInterval at 1000ms
  On each tick:
    1. For each ship:
       a. If ship.path has waypoints, advance toward next waypoint
          - Compute heading to next waypoint using bearing formula
          - speedKmH = ship.speed * 1.852
          - distanceThisTick = speedKmH / 3600  (km per second)
          - Move ship lat/lng that distance toward waypoint
          - If within 0.05km of waypoint, shift to next waypoint
          - If path is empty, ship has arrived → status = 'stopped'
       b. Fuel burn: ship.fuel -= (inWeather ? 0.03 * 1.3 : 0.03)
       c. If fuel <= 0: status = 'stopped', fuel = 0
    2. Emit 'fleet:update' to ALL connected sockets with full ships array

### Bearing formula (include this exactly in simulator.js):
function bearing(lat1, lng1, lat2, lng2) {
  const dLng = (lng2 - lng1) * Math.PI / 180
  const lat1R = lat1 * Math.PI / 180
  const lat2R = lat2 * Math.PI / 180
  const y = Math.sin(dLng) * Math.cos(lat2R)
  const x = Math.cos(lat1R)*Math.sin(lat2R) -
            Math.sin(lat1R)*Math.cos(lat2R)*Math.cos(dLng)
  return (Math.atan2(y, x) * 180 / Math.PI + 360) % 360
}

### Coordinate advance formula:
function moveAlongBearing(lat, lng, bearingDeg, distKm) {
  const R = 6371
  const d = distKm / R
  const b = bearingDeg * Math.PI / 180
  const lat1 = lat * Math.PI / 180
  const lng1 = lng * Math.PI / 180
  const lat2 = Math.asin(Math.sin(lat1)*Math.cos(d) +
                Math.cos(lat1)*Math.sin(d)*Math.cos(b))
  const lng2 = lng1 + Math.atan2(Math.sin(b)*Math.sin(d)*Math.cos(lat1),
                Math.cos(d)-Math.sin(lat1)*Math.sin(lat2))
  return { lat: lat2*180/Math.PI, lng: lng2*180/Math.PI }
}

Do NOT use any library for these two functions. They are self-contained.

## PHASE 2 — ROUTING ENGINE
File: lib/router.js

### computePath(ship, zones)
Simple waypoint deflection algorithm (NOT full A*):

1. Start with a direct line from ship position to destination
2. Generate 8 intermediate waypoints along that line (evenly spaced)
3. For each waypoint:
   - Check if it falls inside any zone polygon using turf.booleanPointInPolygon
   - If inside: deflect the waypoint perpendicular to the line direction by
     0.3 degrees (enough to skirt the zone). Try both left and right; pick
     whichever is inside NAVIGABLE_WATER.
4. Return the array of waypoints (always ends at destination)
5. If destination itself is inside a zone: return [] and set status='stranded'

### How to use Turf correctly (DO NOT deviate):
const turf = require('@turf/turf')

// Point check:
const pt = turf.point([lng, lat])  // NOTE: turf is [lng, lat] not [lat, lng]
const poly = turf.polygon([coordinates])
const inside = turf.booleanPointInPolygon(pt, poly)

// Distance:
const dist = turf.distance(turf.point([lng1,lat1]), turf.point([lng2,lat2]),
             { units: 'kilometers' })

IMPORTANT: Turf uses [longitude, latitude] order everywhere.
Your ship objects use { lat, lng }. Always convert: turf.point([ship.lng, ship.lat])

## PHASE 3 — GEOFENCE + PROXIMITY CHECKS
File: lib/geofence.js

### checkGeofences(ships, zones)
- For each ship, for each zone: booleanPointInPolygon
- If ship is inside a zone AND was not inside last tick:
  emit 'alert:geofence' via the io instance with { shipId, zoneName, severity: 'high' }
  set ship.status = 'rerouting', trigger computePath

### checkProximity(ships)
- Loop all pairs O(n²) — 15 ships = 105 pairs, fine
- turf.distance between each pair
- If < 2km AND pair not already warned:
  emit 'alert:proximity' with { ship1Id, ship2Id, distanceKm }

### Make io accessible to geofence.js:
In server.js: pass io into simulator and geofence modules at startup.
Pattern: geofence.init(io) then call geofence.checkAll(ships, zones) inside tick.

const SNAPSHOT_INTERVAL = 30  // seconds
const MAX_SNAPSHOTS = 120      // 60 minutes worth

let snapshots = []
let tickCount = 0

function maybeSnapshot(ships) {
  tickCount++
  if (tickCount % SNAPSHOT_INTERVAL === 0) {
    snapshots.push({
      timestamp: Date.now(),
      ships: JSON.parse(JSON.stringify(ships))  // deep clone
    })
    if (snapshots.length > MAX_SNAPSHOTS) snapshots.shift()
  }
}

function getSnapshots() { return snapshots }

Export both. Call maybeSnapshot(ships) inside simulator tick.
Expose GET /api/playback in server.js that returns getSnapshots().

### hooks/useFleetSocket.js
import { useEffect, useState, useRef } from 'react'
import { io } from 'socket.io-client'

export function useFleetSocket() {
  const [ships, setShips] = useState([])
  const [alerts, setAlerts] = useState([])
  const [zones, setZones] = useState([])
  const socketRef = useRef(null)

  useEffect(() => {
    const socket = io('http://localhost:3001')
    socketRef.current = socket

    socket.on('fleet:state', setShips)
    socket.on('fleet:update', setShips)
    socket.on('alert:geofence', alert => {
      setAlerts(prev => [...prev, { ...alert, id: Date.now(), acked: false }])
      // Play alert sound:
      new Audio('/alert.mp3').play().catch(() => {})
    })
    socket.on('alert:proximity', alert => {
      setAlerts(prev => [...prev, { ...alert, id: Date.now(), acked: false }])
    })
    socket.on('zone:updated', ({ zones }) => setZones(zones))

    return () => socket.disconnect()
  }, [])

  const sendDirective = (shipId, action, params) =>
    socketRef.current?.emit('directive:send', { shipId, action, params })

  const respondDirective = (shipId, response, message) =>
    socketRef.current?.emit('directive:respond', { shipId, response, message })

  const ackAlert = (alertId) =>
    setAlerts(prev => prev.map(a => a.id === alertId ? {...a, acked: true} : a))

  const addZone = (polygon, name) =>
    socketRef.current?.emit('zone:add', { polygon, name })

  return { ships, alerts, zones, sendDirective, respondDirective, ackAlert, addZone }
}

useEffect(() => {
  const start = { ...lerpPos.current }

  const startTime = performance.now()

## PHASE 8 — AI DISTRESS PARSING
File: pages/api/distress.js

import Anthropic from '@anthropic-ai/sdk'

const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY })

export default async function handler(req, res) {
  if (req.method !== 'POST') return res.status(405).end()
  const { message, shipId } = req.body

  const response = await client.messages.create({
    model: 'claude-haiku-4-5-20251001',
    max_tokens: 256,
    messages: [{
      role: 'user',
      content: `You are a maritime emergency analyst. Extract structured data from this distress message.
      
Distress message: "${message}"

Respond with ONLY valid JSON, no explanation:
{
  "severity": 1-5,
  "incidentType": "fire|medical|mechanical|collision|weather|cargo|unknown",
  "injuries": number or null,
  "damagePct": 0-100 or null,
  "immediateRisk": true|false,
  "summary": "one sentence"
}`
    }]
  })

  try {
    const parsed = JSON.parse(response.content[0].text)
    res.json({ shipId, ...parsed })
  } catch {
    res.json({ shipId, severity: 3, incidentType: 'unknown', summary: message })
  }
}
